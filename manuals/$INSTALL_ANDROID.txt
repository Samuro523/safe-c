
Install Android
===============

0. How to compile for Android
-----------------------------

Developping for Android is much harder than PC/Windows. 
You will get headaches from configuration problems.

In Android you can launch ONLY Java programs.
However the Java program can link with shared libraries (.so) compiled in other languages.

The Safe-C compiler must thus produce a .so shared library.
For example your program MUST be called "libsafec.c" and compile into "libsafec.so"

You compile on PC/Windows, then you use a batch to sign+load the program
into your android phone and run it. That's explained below.

// main.cpp

#include <jni.h>
#include <game-activity/GameActivity.cpp>
#include <game-text-input/gametextinput.cpp>

/*----------------------------------------------------------------------------*/

extern "C" {
#include <dlfcn.h>
#include <game-activity/native_app_glue/android_native_app_glue.c>

/*----------------------------------------------------------------------------*/

const int EVENT_INITIALIZE   =   1;
const int EVENT_RENDER_FRAME =  10;
const int EVENT_TERMINATE    = 999;

static int (*fsafec_event)(android_app *pApp, int event_nr);

#define LOG_TAGS "Safe-C-Logging"
#define LOG(...) ((void)__android_log_print(ANDROID_LOG_FATAL, LOG_TAGS, __VA_ARGS__))

/*----------------------------------------------------------------------------*/

static int init_safec_so ()
{
    const char *lib = "libsafec.so";
    void *hso = dlopen (lib, RTLD_NOW);
    if (hso == NULL)
    {
        LOG ("FATAL ERROR: dlopen(libsafec.so) failed: %s", dlerror());
        return -1;
    }

    const char *func = "safec_event";
    fsafec_event = (int (*)(android_app *pApp, int)) dlsym(hso, func);
    if (fsafec_event == NULL)
    {
        LOG ("FATAL ERROR: dlsym(%s,%s) failed: %s", lib, func, dlerror());
        return -1;
    }

    return 0;
}

/*----------------------------------------------------------------------------*/

void handle_cmd (android_app *pApp, int32_t cmd)
{
    switch (cmd)
    {
        case APP_CMD_INIT_WINDOW:      // A new window is created
            fsafec_event(pApp, EVENT_INITIALIZE);
            pApp->userData = (void*)1;   // now loop downstairs renders frames
            break;


        case APP_CMD_TERM_WINDOW:      // The window is being destroyed.
            if (pApp->userData)   // initialization was done ?
            {
                pApp->userData = (void*)0;   // no longer render frames
                fsafec_event(pApp, EVENT_TERMINATE);
            }
            break;

        default:
            break;
    }
}

/*----------------------------------------------------------------------------*/

// This the main entry point for a native activity

void android_main(struct android_app *pApp)
{
    if (init_safec_so () < 0)
        return;

    pApp->onAppCmd = handle_cmd;

    // This sets up a typical game/event loop. It will run until the app is destroyed.
    for (;;)
    {
        // Process all pending events before running game logic.
        bool done = false;

        while (!done)
        {
            int                 timeout, events, result;
            android_poll_source *pSource;

            timeout = 0;    // 0 is non-blocking.
            result = ALooper_pollOnce (timeout, nullptr, &events, reinterpret_cast<void**>(&pSource));

            switch (result)
            {
                case ALOOPER_POLL_TIMEOUT:
                    [[clang::fallthrough]];
                case ALOOPER_POLL_WAKE:
                    // No events occurred before the timeout or explicit wake. Stop checking for events.
                    done = true;
                    break;

                case ALOOPER_EVENT_ERROR:
                    break;

                case ALOOPER_POLL_CALLBACK:
                    break;

                default:
                    if (pSource)
                        pSource->process(pApp, pSource);
                    break;
            }
        }

        if (pApp->userData)
        {
            // Render a frame
            fsafec_event(pApp, EVENT_RENDER_FRAME);
        }

        if (pApp->destroyRequested)
            break;
    }
}

/*-----------------------------------------------------------------------*/
}  // extern "C"
/*----------------------------------------------------------------------------*/

7. Compile/Execute in Safe-C
----------------------------

a) Android studio must generate a file "app-release.apk" that you put in the current folder.
b) Your Safe-C program MUST be called "libsafec.c" and compile into "libsafec.so"
   (you must never rename the .so file after compilation as the compiler stores its name in the elf format !)

        mk libsafec.c cfg=android.cfg crc %1

   with android.cfg containing :

        [options]
        target          = android   ; intel or android
        memory_model    = 64        ; 32 or 64 (default is 64)
        pointer_check   = yes       ; default is yes
        array_check     = yes       ; default is yes
        assertion_check = yes       ; default is yes

    %1 can be replaced by prod when compiling for production, identical strings are then merged but that compiles slowlier.

c) zipadd.exe is provided in the compiler tools.
    zipadd.exe
       add file in zip archive
       format: zipadd <android_filename_to_add>  <pc_filename_to_add>  <source_zip>  <target_zip>

d) zipalign.exe, apksigner.bat and adb.exe must be downloaded
   from https://developer.android.com/studio/releases/build-tools.html

e) make sure developer mode is enable on your phone (search on the internet how to do that)
f) connect your phone to PC using an USB cable, 
g) use the following batch :

      cls
      mk libsafec.c cfg=android.cfg crc %1
      if %ERRORLEVEL% NEQ 0 goto :ending

      zipadd lib\arm64-v8a\libsafec.so libsafec.so app-release.apk app-release.tmp1
      zipadd assets\libsafec.dbg libsafec.dbg app-release.tmp1 app-release.tmp2

10. Example Safe-C app
----------------------

    // libsafec.c

    from std use exception, draw3d, tracing, android/android;
    use planet_client;

    //---------------------------------------------------------------------
    #begin unsafe
    //---------------------------------------------------------------------

    bool g_init_done;

    //---------------------------------------------------------------------

    // event type is called with (EVENT_INITIALIZE, EVENT_RENDER_FRAME or EVENT_TERMINATE)

    [entry]
    int safec_event (android_app *app, int event_type)
    {
      if (!g_init_done)  // on first run only
      {
        g_init_done = true;

        init_android (app);   // save some app pointers and set current directory (must be done before arming exception handler)

        arm_exception_handler ();

        open_trace ("dummy-trace");

        // init opengl so that dialogs can be displayed
        draw3d . event_3D (app, EVENT_INITIALIZE);

        // init application, init draw3 functions, starting a window thread.
        planet_client . planet_client_main ();

        return 0;   // finish
      }

      // draw3d thread (EVENT_INITIALIZE, EVENT_RENDER_FRAME or EVENT_TERMINATE)
      draw3d . event_3D (app, event_type);

      return 0;
    }

    //---------------------------------------------------------------------
    #end unsafe
    //---------------------------------------------------------------------

11. Java Code to receive a data file
------------------------------------

package com.example.planetsamuro;

import android.content.Intent;
import android.net.Uri;
import android.os.Bundle;
import android.view.View;
import android.widget.TextView;

import com.google.androidgamesdk.GameActivity;

import java.io.File;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.OutputStream;

public class MainActivity extends GameActivity {
    static {
        System.loadLibrary("planetsamuro");
    }

    @Override
    public void onWindowFocusChanged(boolean hasFocus) {
        super.onWindowFocusChanged(hasFocus);

        if (hasFocus) {
            hideSystemUi();
        }
    }

    private void hideSystemUi() {
        View decorView = getWindow().getDecorView();
        decorView.setSystemUiVisibility(
                View.SYSTEM_UI_FLAG_IMMERSIVE_STICKY
                        | View.SYSTEM_UI_FLAG_LAYOUT_STABLE
                        | View.SYSTEM_UI_FLAG_LAYOUT_HIDE_NAVIGATION
                        | View.SYSTEM_UI_FLAG_LAYOUT_FULLSCREEN
                        | View.SYSTEM_UI_FLAG_HIDE_NAVIGATION
                        | View.SYSTEM_UI_FLAG_FULLSCREEN
        );
    }

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        Intent intent = getIntent();
        String action = intent.getAction();
        String type = intent.getType();

        if (Intent.ACTION_SEND.equals(action) && type != null) {
            Uri fileUri = intent.getParcelableExtra(Intent.EXTRA_STREAM);
            if (fileUri != null) {
                saveBinaryFile(fileUri);
            }
        }
    }

    private void saveBinaryFile(Uri uri) {
        try {
            InputStream inputStream = getContentResolver().openInputStream(uri);
            File outFile = new File(getFilesDir(), "INVENTORY.DB.NEW");
            OutputStream outputStream = new FileOutputStream(outFile);

            byte[] buffer = new byte[4096];
            int bytesRead;
            while ((bytesRead = inputStream.read(buffer)) != -1) {
                outputStream.write(buffer, 0, bytesRead);
            }

            inputStream.close();
            outputStream.close();

        } catch (Exception e) {
        }
    }
}
